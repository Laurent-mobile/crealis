Parfait ! Serveo.net est une excellente alternative gratuite qui permet d'avoir plusieurs tunnels simultanément. Voici comment l'utiliser :

Installation et utilisation de Serveo
Serveo ne nécessite aucune installation ! Il utilise SSH directement depuis PowerShell.

Étape 1 : Démarrer le backend localement
powershell
# Terminal 1 - Démarrer le backend
cd C:\MesApplications\CREALIS2\backend
npm run dev
Étape 2 : Démarrer le frontend localement
powershell
# Terminal 2 - Démarrer le frontend
cd C:\MesApplications\CREALIS2\frontend-web
npm run dev
Étape 3 : Créer un tunnel pour le backend (nouveau terminal)
powershell
# Terminal 3 - Tunnel pour le backend sur le port 5000
ssh -R 80:localhost:5000 serveo.net
Important : La première fois, on vous demandera de confirmer la connexion. Tapez yes et Entrée.

Serveo vous donnera une URL comme : https://xxx.serveo.net

📌 COPIEZ CETTE URL pour le backend

Étape 4 : Créer un tunnel pour le frontend (autre nouveau terminal)
powershell
# Terminal 4 - Tunnel pour le frontend sur le port 3001
ssh -R 80:localhost:3001 serveo.net
Serveo vous donnera une autre URL : https://yyy.serveo.net

📌 COPIEZ CETTE URL pour le frontend

Étape 6 : Configurer le frontend pour utiliser l'API distante
Modifiez frontend-web/src/api.js :

javascript
import axios from 'axios';

// Configuration Serveo
// REMPLACEZ par votre URL Serveo du backend
const SERVER_API_URL = 'https://crealis-api.serveo.net/api';

const getApiUrl = () => {
    // Si on est sur l'URL Serveo du frontend
    if (window.location.hostname.includes('serveo.net')) {
        return SERVER_API_URL;
    }
    // En développement local
    return 'http://localhost:5000/api';
};

const API_URL = getApiUrl();
console.log('🔧 API URL:', API_URL);

const api = axios.create({
    baseURL: API_URL,
    headers: {
        'Content-Type': 'application/json',
    },
    timeout: 30000,
});

api.interceptors.request.use(
    (config) => {
        const token = localStorage.getItem('token');
        if (token) {
            config.headers['x-auth-token'] = token;
        }
        return config;
    },
    (error) => Promise.reject(error)
);

export default api;

1. Modifiez frontend-web/src/api.js
javascript
import axios from 'axios';

// Configuration Serveo avec vos URLs
const BACKEND_URL = 'https://ee698ee2319bcb9f-102-178-16-122.serveousercontent.com';
const LOCAL_URL = 'http://localhost:5000';

const getApiUrl = () => {
    const hostname = window.location.hostname;
    
    // Si on est sur l'URL Serveo du frontend
    if (hostname.includes('serveousercontent.com')) {
        return `${BACKEND_URL}/api`;
    }
    
    // En développement local
    if (hostname === 'localhost' || hostname === '127.0.0.1') {
        return 'http://localhost:5000/api';
    }
    
    return 'http://localhost:5000/api';
};

const API_URL = getApiUrl();
console.log('🔧 API URL:', API_URL);

const api = axios.create({
    baseURL: API_URL,
    headers: {
        'Content-Type': 'application/json',
    },
    timeout: 30000,
});

// Intercepteur pour ajouter le token
api.interceptors.request.use(
    (config) => {
        const token = localStorage.getItem('token');
        if (token) {
            config.headers['x-auth-token'] = token;
        }
        return config;
    },
    (error) => {
        return Promise.reject(error);
    }
);

export default api;

Le problème vient du fait que le token est bien stocké mais la redirection ne fonctionne pas à cause de l'URL Serveo. Voici la correction :

Corriger Login.jsx - Problème de redirection
jsx
import React, { useState } from 'react'
import { useNavigate } from 'react-router-dom'
import api from '../api'
import toast from 'react-hot-toast'

function Login() {
  const [email, setEmail] = useState('')
  const [password, setPassword] = useState('')
  const [loading, setLoading] = useState(false)
  const [showPassword, setShowPassword] = useState(false)
  const navigate = useNavigate()

  const handleSubmit = async (e) => {
    e.preventDefault()
    setLoading(true)
    try {
      const response = await api.post('/auth/login', { email, password })
      console.log('Login response:', response.data)
      
      // Stocker le token et l'utilisateur
      localStorage.setItem('token', response.data.token)
      localStorage.setItem('user', JSON.stringify(response.data.user))
      
      // Vérifier que le token est bien stocké
      const token = localStorage.getItem('token')
      console.log('Token stored:', token ? 'Yes' : 'No')
      
      toast.success('Connexion réussie')
      
      // Forcer la redirection
      window.location.href = '/'
      
    } catch (error) {
      console.error('Login error:', error)
      toast.error('Email ou mot de passe incorrect')
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className="login-container">
      <div className="login-bg">
        <div className="login-bg-overlay"></div>
        <div className="login-bg-particles">
          <div className="particle"></div>
          <div className="particle"></div>
          <div className="particle"></div>
          <div className="particle"></div>
          <div className="particle"></div>
        </div>
      </div>

      <div className="login-card">
        <div className="login-header">
          <div className="login-logo">
            <div className="logo-icon-large">🏪</div>
            <div className="logo-text-large">
              <span className="logo-main">CREALIS</span>
              <span className="logo-sub">DASSANDAGA</span>
            </div>
          </div>
          <p className="login-subtitle">Plateforme de gestion intelligente</p>
        </div>

        <form onSubmit={handleSubmit} className="login-form">
          <div className="input-group">
            <div className="input-icon">📧</div>
            <input
              type="email"
              className="login-input"
              placeholder="Adresse email"
              value={email}
              onChange={(e) => setEmail(e.target.value)}
              required
            />
          </div>

          <div className="input-group">
            <div className="input-icon">🔒</div>
            <input
              type={showPassword ? 'text' : 'password'}
              className="login-input"
              placeholder="Mot de passe"
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              required
            />
            <button 
              type="button" 
              className="password-toggle"
              onClick={() => setShowPassword(!showPassword)}
            >
              {showPassword ? '👁️' : '👁️‍🗨️'}
            </button>
          </div>

          <div className="login-options">
            <label className="checkbox-label">
              <input type="checkbox" /> 
              <span>Se souvenir de moi</span>
            </label>
            <a href="#" className="forgot-link">Mot de passe oublié ?</a>
          </div>

          <button 
            type="submit" 
            className="login-btn"
            disabled={loading}
          >
            {loading ? (
              <span className="btn-spinner"></span>
            ) : (
              <>
                <span>Se connecter</span>
                <span className="btn-icon">→</span>
              </>
            )}
          </button>
        </form>

        <div className="login-footer">
          <div className="login-features">
            <div className="feature">
              <span className="feature-icon">📊</span>
              <span>Dashboard en temps réel</span>
            </div>

            </div>
            <div className="feature">
              <span className="feature-icon">📱</span>

            </div>
          </div>
          <p className="login-copyright">
            © 2025 CREALIS-DASSANDAGA - Tous droits réservés
          </p>
        </div>
      </div>
    </div>
  )
}

Corriger App.jsx - Vérification d'authentification
jsx
import React from 'react'
import { BrowserRouter, Routes, Route, Navigate } from 'react-router-dom'
import Dashboard from './pages/Dashboard'
import Merchants from './pages/Merchants'
import Transactions from './pages/Transactions'
import Loyalty from './pages/Loyalty'
import Customers from './pages/Customers'
import Products from './pages/Products'
import Warehouse from './pages/Warehouse'
import POS from './pages/POS'
import Advertising from './pages/Advertising'
import Reports from './pages/Reports'
import Settings from './pages/Settings'
import Login from './pages/Login'

function App() {
  // Vérification plus robuste
  const token = localStorage.getItem('token')
  const isAuthenticated = token !== null && token !== 'undefined' && token !== ''

  return (
    <BrowserRouter>
      <Routes>
        <Route path="/login" element={<Login />} />
        <Route path="/" element={isAuthenticated ? <Dashboard /> : <Navigate to="/login" />} />
        <Route path="/merchants" element={isAuthenticated ? <Merchants /> : <Navigate to="/login" />} />
        <Route path="/transactions" element={isAuthenticated ? <Transactions /> : <Navigate to="/login" />} />
        <Route path="/loyalty" element={isAuthenticated ? <Loyalty /> : <Navigate to="/login" />} />
        <Route path="/customers" element={isAuthenticated ? <Customers /> : <Navigate to="/login" />} />
        <Route path="/products" element={isAuthenticated ? <Products /> : <Navigate to="/login" />} />
        <Route path="/warehouse" element={isAuthenticated ? <Warehouse /> : <Navigate to="/login" />} />
        <Route path="/pos" element={isAuthenticated ? <POS /> : <Navigate to="/login" />} />
        <Route path="/advertising" element={isAuthenticated ? <Advertising /> : <Navigate to="/login" />} />
        <Route path="/reports" element={isAuthenticated ? <Reports /> : <Navigate to="/login" />} />
        <Route path="/settings" element={isAuthenticated ? <Settings /> : <Navigate to="/login" />} />
      </Routes>
    </BrowserRouter>
  )
}

export default App
